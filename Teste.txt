// PROXIMO ----------------
app.get("/proximo", (req, res) => {
    const proxima = db.prepare(`
        SELECT id, senha
        FROM senhas
        WHERE status = 'aguardando'
        ORDER BY
            CASE
                WHEN tipo = 'preferencial' THEN 0
                ELSE 1
            END,
            id ASC
        LIMIT 1
    `).get();

    if (!proxima) {
        return res.json({});
    }

    db.prepare(`
        UPDATE senhas
        SET status = 'chamada'
        WHERE id = ?
    `).run(proxima.id);

    res.json({ senha: proxima.senha });
});
